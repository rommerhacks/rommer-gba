# elite-4-randomizer: after a new game, Lorelei's battle starts by itself (only A presses for her
# intro text) right after START -> NEW GAME on the title screen, against TRAINER_ELITE_FOUR_LORELEI, with the level-60 random party.

state {project}/re/states/title.ss0
until {e4r_new_game} 6000 START,A
# CB2_InitBattle
until 0x0800FDB0 3000 A
expect32 0x03005008 0x020255A0 gSaveBlock1Ptr guard
expect32 0x020255A0+0x3A4C 0xBFBFCCC1 rivalName = GREE (e4r_skip_intro default)
expect16 0x020255A0+0x3A50 0xFFC8 rivalName = N, EOS
expect16 0x020386AE 410 gTrainerBattleOpponent_A = TRAINER_ELITE_FOUR_LORELEI
expect8 0x02024029 6 gPlayerPartyCount = 6
frames 600
# gBattleMons[0] (player's lead) and [1] (Lorelei's Dewgong)
expect8 0x02023BE4+0x2A 60 player lead level = 60 (loadout)
expect16 0x02023BE4+0x58 87 opponent lead = DEWGONG
