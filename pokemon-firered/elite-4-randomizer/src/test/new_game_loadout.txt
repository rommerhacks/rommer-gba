# elite-4-randomizer: what a new game sets up before the field loads.
# Mashes START/A from the title screen (empty save: START -> NEW GAME, which now skips Oak's intro)
# and stops right after NewGameInitData's call to WarpToPlayersRoom (now e4r_new_game). The XOR key is still 0
# here, so bag quantities are plain. Pointers are deterministic for this state + input (guards).

# hook: StartNewGameScene -> long call e4r_skip_intro (same shape as below)
expect32 0x0812EBA4 0x4802B500 hook: push {lr}; ldr r0,[pc,#8]
expect32 0x0812EBB0 {e4r_skip_intro}+1 hook literal = e4r_skip_intro|1
# hook: WarpToPlayersRoom -> push {lr}; ldr r0,=e4r_new_game; bl _call_via_r0; pop {r0}; bx r0
expect32 0x08054A0C 0x4802B500 hook: push {lr}; ldr r0,[pc,#8]
expect32 0x08054A18 {e4r_new_game}+1 hook literal = e4r_new_game|1
# Lorelei's room map scripts: entry 5 = type 2 (ON_FRAME_TABLE) + unaligned pointer at 0x081625F9
expect32 0x081625F8 0xEB134802 ON_FRAME table -> e4r_lorelei_on_frame (0x08EB1348), low bytes
expect8 0x081625FC 0x08 ON_FRAME table pointer, high byte

state {project}/re/states/title.ss0
until 0x08054B42 6000 START,A
expect32 0x03005008 0x02025584 gSaveBlock1Ptr guard
expect8 0x02024029 6 gPlayerPartyCount = 6
expect32 0x0300500C 0x020245E0 gSaveBlock2Ptr guard

# e4r_skip_intro defaults: boy "RED" (SaveBlock2+0, +8). The rival name is checked in
# lorelei_autostart: NewGameInitData only restores it after this point.
expect32 0x020245E0 0xFFBEBFCC playerName = RED
expect8 0x020245E0+0x8 0 playerGender = MALE

# bag items pocket (SaveBlock1+0x310): item id, quantity (loadout edit set)
expect32 0x02025584+0x310 0x00050013 FULL RESTORE x5
expect32 0x02025584+0x314 0x00030019 MAX REVIVE x3
expect32 0x02025584+0x318 0x00030017 FULL HEAL x3
expect32 0x02025584+0x31C 0x00020025 MAX ELIXIR x2
expect16 0x02025584+0x320 0 no fifth item

# flags (SaveBlock1+0xEE0): all 8 badges (0x820-0x827), POKEMON_GET + POKEDEX_GET (0x828, 0x829)
expect8 0x02025584+0xEE0+0x104 0xFF badges 1-8
expect8 0x02025584+0xEE0+0x105 0x03 FLAG_SYS_POKEMON_GET | FLAG_SYS_POKEDEX_GET
# VAR_MAP_SCENE_POKEMON_LEAGUE (0x4068) = 2: our auto-start entry in Lorelei's ON_FRAME table
expect16 0x02025584+0x1000+0xD0 2 VAR_MAP_SCENE_POKEMON_LEAGUE = autostart

# warp target: Lorelei's room (group 1, map 75), warp -1, (6,6) facing Lorelei at (6,5)
expect32 0x02025584+0x4 0x00FF4B01 location = PokemonLeague_LoreleisRoom, warp -1
expect32 0x02025584+0x8 0x00060006 location x=6 y=6
# respawn = HEAL_LOCATION_INDIGO_PLATEAU (Indigo Plateau exterior 3.9, (11,7))
expect32 0x02025584+0x1C 0x00FF0903 lastHealLocation = Indigo Plateau
expect32 0x02025584+0x20 0x0007000B lastHealLocation x=11 y=7
